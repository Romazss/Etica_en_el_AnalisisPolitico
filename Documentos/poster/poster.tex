% COLUMNA 1: POSTURA ÉTICA Y CONTEXTO
\section*{\textcolor{redaccent}{POSTURA ETICA CENTRAL}}
{\fontsize{18}{22}\selectfont \textbf{\textcolor{darkgray}{Los algoritmos de redes sociales fomentan activamente la polarizacion politica, erosionando la calidad democratica en Chile.}}}

\vspace{0.3cm}

\subsection*{\textcolor{blueaccent}{Tres consecuencias negativas:}}
\begin{itemize}
    \item[\textbf{\textcolor{redaccent}{X}}] \textbf{Limitacion del dialogo democratico}: ``Burbujas de filtro'' impiden exposicion a posturas diversas
    \item[\textbf{\textcolor{redaccent}{X}}] \textbf{Facilitacion de desinformacion}: Algoritmos priorizan engagement sobre veracidad
    \item[\textbf{\textcolor{redaccent}{X}}] \textbf{Reproduccion de sesgos}: Sistemas de IA aprenden y refuerzan prejuicios
\end{itemize}

\vspace{0.5cm}

\section*{\textcolor{blueaccent}{ANALISIS SOCIOTECNICO}}

\subsection*{\textbf{Mecanismos de Recomendación:}}
\begin{enumerate}
    \item Extracción de datos del perfil de usuario
    \item Aprendizaje automático con patrones de comportamiento
    \item Predicción de engagement (tiempo, interacciones, comparticiones)
    \item Personalización de feed según preferencias inferidas
    \item \textbf{\textcolor{redaccent}{Resultado: Burbuja de Filtro}}
\end{enumerate}

\subsection*{\textbf{Datos Clave del Ecosistema Digital Chileno:}}
\begin{itemize}
    \item 17,85 millones de usuarios en redes sociales
    \item 92\% de la población conectada
    \item 20,000 bots detectados en elección 2021
    \item 72,9\% vio desinformación en Twitter (2022)
\end{itemize}

% COLUMNA 2: CASOS CHILENOS Y DESINFORMACIÓN
\section*{\textcolor{darkgray}{CASOS CHILENOS (2021-2025)}}

\subsection*{\textbf{\textcolor{redaccent}{Eleccion Presidencial 2021}}}
\textbf{Protagonistas:} Boric vs. Kast | \textbf{Plataforma clave:} TikTok \\
\textbf{Impacto:} Segmentación generacional extrema, 20,000 bots operando

\subsection*{\textbf{\textcolor{redaccent}{Plebiscitos 2022-2023}}}
\textbf{Campaña sistematica}: 5 meses (3 fuera del período legal) \\
\textbf{Estrategia}: Apelar al miedo, no al debate \\
\textbf{Temas manipulados}: Pensiones (``Con mi plata''), vivienda, salud, educacion

\vspace{0.3cm}
\textbf{\textcolor{blueaccent}{Punto de inflexion}}: 1 de abril 2022 \\
Titular falso de Bernardo Fontaine: ``Los trabajadores ya no seran dueños de sus ahorros previsionales'' \\
\textit{Coincidio exactamente con cuando encuesta CADEM mostro rechazo superando apruebo}

\subsection*{\textbf{\textcolor{redaccent}{Campañas 2024-2025}}}
\textbf{Red de bots contra candidatas}: Evelyn Matei (Alzheimer falso) y Janet Jara (videos adulterados) \\
\textbf{Origenes}: Asociacion de AFP financiando operaciones clandestinas \\
\textbf{Herramienta}: ``Cuentas madre'' del Partido Republicano (76\% ataques a mujeres)

\vspace{0.5cm}

\section*{\textcolor{darkgray}{DESINFORMACION DOCUMENTADA}}

\subsection*{\textbf{Desde plataformas académicas:}}
\begin{itemize}
    \item \textbf{72,9\%} de encuestados vio fake news en Twitter (plebiscito 2022)
    \item \textbf{49\%} de latinoamericanos se topa con noticias falsas \textbf{diariamente}
    \item \textbf{74,4\%} de chilenos apoya sancionar difusión de fake news
    \item \textbf{75,7\%} considera desinformación amenaza a democracia
\end{itemize}

\subsection*{\textbf{WhatsApp: nueva arena de desinformación}}
\begin{itemize}
    \item 25\% de mensajes en grupos políticos públicos contienen desinformación
    \item 100\% de desinformación sobre plebiscito era pro-rechazo
    \item Plataforma cerrada dificulta regulación y fact-checking
\end{itemize}

% COLUMNA 3: ACTORES, ANÁLISIS ÉTICO Y PROPUESTAS
\section*{\textcolor{darkgray}{ACTORES CLAVE Y DINAMICAS DE PODER}}

\textbf{Asimetría de información:} Usuarios desconocen cómo algoritmos operan \\
\textbf{Concentración de poder:} Meta, X, TikTok dominan infraestructura política \\
\textbf{Falta de transparencia:} ``Caja negra'' algorítmica sin auditoría independiente \\
\textbf{Mercantilización:} Atención política como producto vendible

\vspace{0.3cm}
\textbf{Grupos económicos transnacionales:} AFP (Generali, MetLife, Prudential, Principal) financiando campañas encubiertas

\vspace{0.5cm}

\section*{\textcolor{greenaccent}{ANALISIS ETICO Y RIESGOS}}

\subsection*{\textbf{Riesgos Identificados:}}
\begin{itemize}
    \item \textcolor{redaccent}{\textbf{Erosion democratica}}: Imposibilidad de consenso en hechos
    \item \textcolor{redaccent}{\textbf{Manipulacion selectiva}}: Micro-targeting a grupos vulnerables
    \item \textcolor{redaccent}{\textbf{Polarizacion subjetiva}}: Izquierda cree 73\% derecha vs aborto; realidad: 40\%
    \item \textcolor{redaccent}{\textbf{Radicalizacion}}: Exposicion gradual a contenido extremista
\end{itemize}

\subsection*{\textbf{Vínculo con Polarización Subjetiva:}}
``El problema no es lo que creemos, sino lo mal que entendemos lo que creen los demás'' --- Tercer Estudio Nacional de Polarizaciones

\vspace{0.5cm}

\section*{\textcolor{greenaccent}{PROPUESTAS ETICAS VIABLES}}

\begin{enumerate}
    \item \textbf{Regulación de transparencia algorítmica}
        \begin{itemize}
            \item Auditorías independientes de sistemas de recomendación
            \item Derecho del usuario a entender por qué ve contenido
            \item Modelo europeo (DSA): multas hasta 10\% facturación global
        \end{itemize}
    
    \item \textbf{Alfabetización digital crítica}
        \begin{itemize}
            \item ``Vacuna cognitiva'': juegos (Bad News, Harmony Square)
            \item Mensajes de precisión: preguntar si titular es preciso antes de compartir
            \item Educación sobre tácticas de manipulación psicológica
        \end{itemize}
    
    \item \textbf{Diseño ético de algoritmos}
        \begin{itemize}
            \item Priorizar veracidad sobre engagement
            \item Auditorías de sesgo pre-lanzamiento
            \item Diversidad en equipos de diseño
        \end{itemize}
    
    \item \textbf{Regulación de campañas políticas}
        \begin{itemize}
            \item Registro público de publicidad política
            \item Límites a micro-targeting electoral
            \item Trazabilidad de financiamiento clandestino
        \end{itemize}
\end{enumerate}
